public class BinarySearchFirst{
    public static int binarySearchFirstOccurrence(int[] a, int key){
        int l = 0;
        int r = a.length -1;
        int res = -1;
        while(l <= r){
            int mid = l +(r-l)/2;
            if(a[mid] == key){
                result = mid;
                r = mid - 1;
            }else if(a[mid] > result){
                r = mid - 1;
            }else{
                l = mid + 1;
            }
        }
        return result;
    }
}
